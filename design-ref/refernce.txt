<script>
  (function initCarDriveScreen() {
    let isPlaying = true;
    let isSaved = false;
    let currentSpeed = 1.2;
    const speeds = [1.0, 1.2, 1.5, 2.0];
    let touchStartX = 0;
    let touchStartY = 0;
    let touchStartTime = 0;
    let lastTapTime = 0;

    const gesturePad = document.getElementById('gesturePad');
    const feedbackOverlay = document.getElementById('gestureFeedback');
    const feedbackBadge = document.getElementById('feedbackBadge');
    const feedbackIcon = document.getElementById('feedbackIcon');
    const feedbackText = document.getElementById('feedbackText');
    const progressBar = document.getElementById('progressBar');
    const currentTimeEl = document.getElementById('currentTime');

    const btnBookmark = document.getElementById('btnBookmark');
    const bookmarkIcon = document.getElementById('bookmarkIcon');
    const btnVoice = document.getElementById('btnVoice');
    const voiceModal = document.getElementById('voiceModal');
    const closeVoice = document.getElementById('closeVoice');
    const btnNext = document.getElementById('btnNext');
    const btnSpeed = document.getElementById('btnSpeed');

    // Trigger visual high-contrast haptic response
    function showGestureCue(icon, text) {
      if (!feedbackBadge || !feedbackOverlay) return;
      feedbackIcon.textContent = icon;
      feedbackText.textContent = text;

      feedbackOverlay.classList.remove('bg-primary-container/0');
      feedbackOverlay.classList.add('bg-primary-container/20');
      feedbackBadge.classList.remove('opacity-0', 'scale-90');
      feedbackBadge.classList.add('opacity-100', 'scale-100');

      setTimeout(() => {
        feedbackOverlay.classList.remove('bg-primary-container/20');
        feedbackOverlay.classList.add('bg-primary-container/0');
        feedbackBadge.classList.remove('opacity-100', 'scale-100');
        feedbackBadge.classList.add('opacity-0', 'scale-90');
      }, 350);
    }

    function togglePlayPause() {
      isPlaying = !isPlaying;
      if (isPlaying) {
        showGestureCue('play_arrow', 'Playing');
      } else {
        showGestureCue('pause', 'Paused');
      }
    }

    function skipForward() {
      showGestureCue('fast_forward', '+30s Skip');
      progressBar.style.width = '68%';
      currentTimeEl.textContent = '01:54';
    }

    function replayBack() {
      showGestureCue('replay_10', '-15s Replay');
      progressBar.style.width = '42%';
      currentTimeEl.textContent = '01:09';
    }

    function toggleBookmark() {
      isSaved = !isSaved;
      if (isSaved) {
        bookmarkIcon.textContent = 'bookmark_added';
        bookmarkIcon.classList.add('text-primary-container');
        bookmarkIcon.style.fontVariationSettings = "'FILL' 1";
        showGestureCue('bookmark', 'Blip Saved');
      } else {
        bookmarkIcon.textContent = 'bookmark';
        bookmarkIcon.classList.remove('text-primary-container');
        bookmarkIcon.style.fontVariationSettings = "'FILL' 0";
        showGestureCue('bookmark_border', 'Removed');
      }
    }

    // Touch gesture engine on giant pad
    if (gesturePad) {
      gesturePad.addEventListener('touchstart', (e) => {
        const touch = e.touches[0];
        touchStartX = touch.clientX;
        touchStartY = touch.clientY;
        touchStartTime = Date.now();
      }, { passive: true });

      gesturePad.addEventListener('touchend', (e) => {
        const touch = e.changedTouches[0];
        const deltaX = touch.clientX - touchStartX;
        const deltaY = touch.clientY - touchStartY;
        const elapsedTime = Date.now() - touchStartTime;
        const now = Date.now();

        // Horizontal Swipe detection
        if (Math.abs(deltaX) > 40 && Math.abs(deltaY) < 60 && elapsedTime < 450) {
          if (deltaX > 0) {
            skipForward();
          } else {
            replayBack();
          }
          return;
        }

        // Tap or Double Tap
        if (Math.abs(deltaX) < 15 && Math.abs(deltaY) < 15) {
          if (now - lastTapTime < 280) {
            // Double tap detected -> Save/Bookmark
            toggleBookmark();
            lastTapTime = 0;
          } else {
            // Single tap detected -> Toggle Play/Pause
            lastTapTime = now;
            setTimeout(() => {
              if (lastTapTime === now) {
                togglePlayPause();
              }
            }, 290);
          }
        }
      }, { passive: true });

      // Fallback click for desktop browser testing
      gesturePad.addEventListener('click', (e) => {
        if ('ontouchstart' in window) return; // avoid double firing on touch devices
        togglePlayPause();
      });
    }

    // Pedals Handlers
    if (btnBookmark) {
      btnBookmark.addEventListener('click', (e) => {
        e.stopPropagation();
        toggleBookmark();
      });
    }

    if (btnNext) {
      btnNext.addEventListener('click', (e) => {
        e.stopPropagation();
        showGestureCue('skip_next', 'Next Blip');
        progressBar.style.width = '0%';
        currentTimeEl.textContent = '00:00';
      });
    }

    if (btnSpeed) {
      btnSpeed.addEventListener('click', (e) => {
        e.stopPropagation();
        const nextIdx = (speeds.indexOf(currentSpeed) + 1) % speeds.length;
        currentSpeed = speeds[nextIdx];
        btnSpeed.textContent = currentSpeed.toFixed(1) + 'x';
        showGestureCue('speed', currentSpeed.toFixed(1) + 'x Speed');
      });
    }

    // Voice Modal Controls
    if (btnVoice && voiceModal) {
      btnVoice.addEventListener('click', (e) => {
        e.stopPropagation();
        voiceModal.classList.remove('hidden');
      });
    }

    if (closeVoice && voiceModal) {
      closeVoice.addEventListener('click', (e) => {
        e.stopPropagation();
        voiceModal.classList.add('hidden');
      });
    }
  })();
</script></main></body></html>}
